import numpy as np
import matplotlib.pyplot as plt

# -----------------------------
# Multi-Armed Bandit Settings
# -----------------------------

np.random.seed(42)

# True reward probabilities of the 5 arms
# These probabilities are unknown to the agent
true_probabilities = [0.10, 0.30, 0.50, 0.70, 0.90]

n_arms = 5
n_trials = 1000

# Epsilon values to compare
epsilons = [0.01, 0.1, 0.5]


def run_bandit(epsilon):
    # Estimated value of each arm
    estimated_values = np.zeros(n_arms)

    # Number of times each arm has been selected
    arm_counts = np.zeros(n_arms)

    # Store rewards
    rewards = []

    for trial in range(n_trials):

        # Exploration or exploitation
        if np.random.random() < epsilon:
            # Exploration: choose a random arm
            arm = np.random.randint(n_arms)
        else:
            # Exploitation: choose the arm with highest estimated reward
            arm = np.argmax(estimated_values)

        # Generate reward
        # Reward is 1 with probability of the selected arm,
        # otherwise reward is 0
        reward = np.random.random() < true_probabilities[arm]
        reward = int(reward)

        # Record reward
        rewards.append(reward)

        # Update count
        arm_counts[arm] += 1

        # Update estimated value using incremental average
        estimated_values[arm] += (
            reward - estimated_values[arm]
        ) / arm_counts[arm]

    # Calculate cumulative reward
    cumulative_reward = np.cumsum(rewards)

    # Calculate average reward
    average_reward = cumulative_reward / np.arange(1, n_trials + 1)

    return rewards, cumulative_reward, average_reward, arm_counts, estimated_values


# --------------------------------
# Run the experiment
# --------------------------------

results = {}

for epsilon in epsilons:
    results[epsilon] = run_bandit(epsilon)

    rewards, cumulative_reward, average_reward, arm_counts, estimated_values = results[epsilon]

    print("\nEpsilon =", epsilon)
    print("Total Reward:", sum(rewards))
    print("Average Reward:", np.mean(rewards))
    print("Arm Selection Counts:", arm_counts)
    print("Estimated Probabilities:", estimated_values)
    print("Most Selected Arm:", np.argmax(arm_counts) + 1)


# --------------------------------
# Plot Cumulative Reward
# --------------------------------

plt.figure(figsize=(10, 6))

for epsilon in epsilons:
    cumulative_reward = results[epsilon][1]
    plt.plot(cumulative_reward, label=f"ε = {epsilon}")

plt.xlabel("Trial")
plt.ylabel("Cumulative Reward")
plt.title("Cumulative Reward for Different Epsilon Values")
plt.legend()
plt.grid(True)
plt.show()


# --------------------------------
# Plot Average Reward
# --------------------------------

plt.figure(figsize=(10, 6))

for epsilon in epsilons:
    average_reward = results[epsilon][2]
    plt.plot(average_reward, label=f"ε = {epsilon}")

plt.xlabel("Trial")
plt.ylabel("Average Reward")
plt.title("Average Reward for Different Epsilon Values")
plt.legend()
plt.grid(True)
plt.show()
